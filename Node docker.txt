index.js

const express = require('express');
const MySQL = require('mysql2/promise');

const app = express();

const datosDb = {
    host: process.env.DB_HOST,
    database: process.env.DB_NAME,
    user: process.env.DB_USER,
    password: process.env.DB_PASSWORD
};

app.get('/', async (solicitud, respuesta) => {
    const conexion = await MySQL.createConnection(datosDb);
    const [filas] = await conexion.query('select * from nombres');

    let lista = '';
    filas.forEach((fila) => {
        lista += `<li>${fila.id}: ${fila.nombre}</li>`;
    });

    const pagina = `<ul>${lista}</ul>`;
    respuesta.send(pagina);
    await conexion.end();
});

app.listen(3000, '0.0.0.0', () => {
    console.log('Corriendo');
});
